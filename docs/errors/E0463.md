# E0463: An interface both derived and implemented

A type gets an interface one way: derived from its blocks, with `with I`
or `impl I for T = derive;`, or written by hand in `impl I for T { … }`.
Doing both leaves two answers for every item.

```lugha
import std.json;

struct Point { x: i64, y: i64 } with json.Json // error: E0463

impl json.Json for Point {
    fun to_json(self): json.Value = json.Value.Text("({self.x}, {self.y})");
    fun from_json(v: json.Value, path: string): Result<Point, json.DecodeError> =
        Ok(Point { x: 0, y: 0 });
}

fun main() {}
```

To customise, keep the `impl` and drop the interface from `with`:

```lugha
import std.json;

struct Point { x: i64, y: i64 }

impl json.Json for Point {
    fun to_json(self): json.Value = json.Value.Text("({self.x}, {self.y})");
    fun from_json(v: json.Value, path: string): Result<Point, json.DecodeError> =
        Ok(Point { x: 0, y: 0 });
}

fun main() {}
```
